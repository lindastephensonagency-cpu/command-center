      }

      function renderDate() {
        const now = new Date();
        const hour = now.getHours();

        $("#dayGreeting").textContent =
          hour < 12
            ? "morning"
            : hour < 18
              ? "afternoon"
              : "evening";

        $("#todayDate").textContent = now.toLocaleDateString(undefined, {
          weekday: "long",
          month: "long",
          day: "numeric",
          year: "numeric"
        });
      }

      function renderMetrics() {
        $$("[data-metric-display]").forEach(element => {
          element.textContent =
            Number(state.metrics[element.dataset.metricDisplay] || 0).toLocaleString();
        });

        $("#growthMetrics").innerHTML = metricDefinitions
          .map(([key, label, target]) => {
            const value = Number(state.metrics[key] || 0);

            const onTarget =
              key === "followups"
                ? value <= target
                : value >= target;

            return `
              <article class="panel metric-card">
                <div style="display:flex;justify-content:space-between;gap:10px">
                  <div class="metric-label">${escapeHtml(label)}</div>
                  <span class="pill ${onTarget ? "" : "watch"}">
                    ${onTarget ? "On target" : "Watch"}
                  </span>
                </div>

                <div class="metric-value">${value.toLocaleString()}</div>

                <div class="metric-meta">
                  <span>Target ${target}</span>
                  <span>This week</span>
                </div>
              </article>
            `;
          })
          .join("");

        const fieldMap = {
          opportunities: "metricOpportunities",
          contacted: "metricContacted",
          scheduled: "metricScheduled",
          completed: "metricCompleted",
          applications: "metricApplications",
          followups: "metricFollowups",
          referrals: "metricReferrals",
          events: "metricEvents"
        };

        Object.entries(fieldMap).forEach(([key, id]) => {
          $("#" + id).value = state.metrics[key] ?? 0;
        });
      }

      function renderActions() {
        const list = $("#actionList");
        const priorities = $("#todayPriorityList");

        const ordered = [...state.actions].sort((a, b) =>
          Number(a.done) - Number(b.done) ||
          String(a.due).localeCompare(String(b.due))
        );

        list.innerHTML = ordered.length
          ? ordered.map(action => `
              <div class="list-row ${action.done ? "completed" : ""}">
                <div>
                  <div class="row-title">${escapeHtml(action.title)}</div>
                  <div class="row-meta">
                    ${escapeHtml(action.lane)} · ${formatDate(action.due)}
                  </div>
                </div>

                <div class="row-actions">
                  <button
                    class="button small-button"
                    type="button"
                    data-action-toggle="${action.id}"
                  >
                    ${action.done ? "Reopen" : "Done"}
                  </button>

                  <button
                    class="button small-button danger"
                    type="button"
                    data-action-delete="${action.id}"
                  >
                    Remove
                  </button>
                </div>
              </div>
            `).join("")
          : '<div class="empty-state">No actions yet. Add the week’s next concrete move.</div>';

        const active = ordered
          .filter(action => !action.done)
          .slice(0, 3);

        priorities.innerHTML = active.length
          ? active.map(action => `
              <div class="list-row">
                <div>
                  <div class="row-title">${escapeHtml(action.title)}</div>
                  <div class="row-meta">
                    ${escapeHtml(action.lane)} · ${formatDate(action.due)}
                  </div>
                </div>
                <span class="pill">Next</span>
              </div>
            `).join("")
          : '<div class="empty-state">Your top three open actions will appear here.</div>';
      }

      function renderPartners() {
        const list = $("#partnerList");

        list.innerHTML = state.partners.length
          ? state.partners.map(partner => `
              <div class="list-row">
                <div>
                  <div class="row-title">${escapeHtml(partner.name)}</div>
                  <div class="row-meta">
                    ${escapeHtml(partner.next)} · ${formatDate(partner.due)}
                  </div>
                </div>

                <button
                  class="button small-button danger"
                  type="button"
                  data-partner-delete="${partner.id}"
                >
                  Remove
                </button>
              </div>
            `).join("")
          : '<div class="empty-state">No partner follow-ups yet.</div>';
      }

      function renderReview() {
        $$("[data-review-check]").forEach(input => {
          input.checked = Boolean(
            state.reviewChecks[input.dataset.reviewCheck]
          );
        });

        while (state.commitments.length < 3) {
          state.commitments.push({
            text: "",
            due: ""
          });
        }

        state.commitments.forEach((commitment, index) => {
          $("#commitment" + (index + 1)).value =
            commitment.text || "";

          $("#commitment" + (index + 1) + "Due").value =
            commitment.due || getWeekEnd();
        });
      }

      function renderLis() {
        $("#leadFirstName").value = state.lis.firstName || "";

        updateLeadName();

        $$("[data-lis-question]").forEach(input => {
          input.checked = Boolean(
            state.lis[input.dataset.lisQuestion]
          );
        });
      }

      function updateLeadName() {
        const raw = $("#leadFirstName").value.trim();
        const displayName = raw || "[First Name]";

        $$(".lead-name").forEach(element => {
          element.textContent = displayName;
        });

        state.lis.firstName = raw;
        saveState();
      }

      function formatTimer(totalSeconds) {
        const hours = Math.floor(totalSeconds / 3600);
        const minutes = Math.floor((totalSeconds % 3600) / 60);
        const seconds = totalSeconds % 60;

        const base =
          String(minutes).padStart(2, "0") +
          ":" +
          String(seconds).padStart(2, "0");

        return hours
          ? String(hours).padStart(2, "0") + ":" + base
          : base;
      }

      function updateTimer() {
        $("#timerDisplay").textContent = formatTimer(timerSeconds);
      }

      function toggleTimer() {
        const button = $("#timerToggle");

        if (timerHandle) {
          window.clearInterval(timerHandle);
          timerHandle = null;

          button.textContent = "▶ Resume call";
          button.setAttribute("aria-pressed", "false");
        } else {
          timerHandle = window.setInterval(() => {
            timerSeconds += 1;
            updateTimer();
          }, 1000);

          button.textContent = "Ⅱ Pause call";
          button.setAttribute("aria-pressed", "true");
        }
      }

      function resetTimer() {
        window.clearInterval(timerHandle);

        timerHandle = null;
        timerSeconds = 0;

        updateTimer();

        $("#timerToggle").textContent = "▶ Start call";
        $("#timerToggle").setAttribute("aria-pressed", "false");
      }

      function resetCallTool() {
        resetTimer();

        state.lis = {
          firstName: "",
          q1: false,
          q2: false,
          q3: false
        };

        saveState("Call workspace cleared for a fresh call.");
        renderLis();

        $$("#scriptSections details").forEach((details, index) => {
          details.open = index === 0;
        });

        $("#leadFirstName").focus();
      }

      function wireEvents() {
        window.addEventListener("hashchange", route);

        $("#mobileMenu").addEventListener("click", () => {
          const sidebar = $("#sidebar");

          sidebar.classList.toggle("open");

          $("#mobileMenu").setAttribute(
            "aria-expanded",
            String(sidebar.classList.contains("open"))
          );
        });

        $("#leadFirstName").addEventListener("input", updateLeadName);

        $$("[data-lis-question]").forEach(input => {
          input.addEventListener("change", event => {
            state.lis[event.target.dataset.lisQuestion] =
              event.target.checked;

            saveState();
          });
        });

        $("#timerToggle").addEventListener("click", toggleTimer);
        $("#timerReset").addEventListener("click", resetTimer);
        $("#resetCallTool").addEventListener("click", resetCallTool);

        $("#expandScripts").addEventListener("click", event => {
          const sections = $$("#scriptSections details");

          const shouldOpen = sections.some(details => !details.open);

          sections.forEach(details => {
            details.open = shouldOpen;
          });

          event.currentTarget.textContent =
            shouldOpen ? "Close all" : "Open all";
        });

        $("#actionForm").addEventListener("submit", event => {
          event.preventDefault();

          const title = $("#actionTitle").value.trim();

          if (!title) {
            return;
          }

          state.actions.push({
            id: crypto.randomUUID(),
            title,
            lane: $("#actionLane").value,
            due: $("#actionDue").value,
            done: false
          });

          saveState("Action added.");
          event.currentTarget.reset();
          renderActions();
        });

        $("#actionList").addEventListener("click", event => {
          const toggle = event.target.closest("[data-action-toggle]");
          const remove = event.target.closest("[data-action-delete]");

          if (toggle) {
            const action = state.actions.find(
              item => item.id === toggle.dataset.actionToggle
            );

            if (action) {
              action.done = !action.done;
            }
          }

          if (remove) {
            state.actions = state.actions.filter(
              item => item.id !== remove.dataset.actionDelete
            );
          }

          if (toggle || remove) {
            saveState("Actions updated.");
            renderActions();
          }
        });

        $("#metricsForm").addEventListener("submit", event => {
          event.preventDefault();

          const fieldMap = {
            opportunities: "metricOpportunities",
            contacted: "metricContacted",
            scheduled: "metricScheduled",
            completed: "metricCompleted",
            applications: "metricApplications",
            followups: "metricFollowups",
            referrals: "metricReferrals",
            events: "metricEvents"
          };

          Object.entries(fieldMap).forEach(([key, id]) => {
            state.metrics[key] = Math.max(
              0,
              Number($("#" + id).value) || 0
            );
          });

          saveState("Weekly totals saved in this browser.");
          renderMetrics();
        });

        $$("[data-review-check]").forEach(input => {
          input.addEventListener("change", event => {
            state.reviewChecks[event.target.dataset.reviewCheck] =
              event.target.checked;

            saveState();
          });
        });

        $("#reviewForm").addEventListener("submit", event => {
          event.preventDefault();

          state.commitments = [1, 2, 3].map(number => ({
            text: $("#commitment" + number).value.trim(),
            due: $("#commitment" + number + "Due").value
          }));

          saveState("Weekly review draft saved.");
        });

        $("#clearReview").addEventListener("click", () => {
          state.reviewChecks = {};
          state.commitments =
            structuredClone(defaultState.commitments);

          saveState("Review cleared for a fresh week.");
          renderReview();
        });

        $("#partnerForm").addEventListener("submit", event => {
          event.preventDefault();

          const name = $("#partnerName").value.trim();
          const next = $("#partnerNext").value.trim();

          if (!name || !next) {
            return;
          }

          state.partners.push({
            id: crypto.randomUUID(),
            name,
            next,
            due: $("#partnerDue").value
          });

          saveState("Partner follow-up added.");
          event.currentTarget.reset();
          renderPartners();
        });

        $("#partnerList").addEventListener("click", event => {
          const remove = event.target.closest("[data-partner-delete]");

          if (!remove) {
            return;
          }

          state.partners = state.partners.filter(
            item => item.id !== remove.dataset.partnerDelete
          );

          saveState("Partner follow-up removed.");
          renderPartners();
        });
      }

      function initialize() {
        renderDate();
        renderMetrics();
        renderActions();
        renderPartners();
        renderReview();
        renderLis();
        updateTimer();
        wireEvents();
        route();
      }

      initialize();
    })();
  </script>
